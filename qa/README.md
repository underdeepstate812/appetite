# QA-скрипты (Playwright, Python)

Запуск при поднятых dev-серверах (backend :3000, mobile-web :8081, web :5174, admin :5173):

```bash
pip install playwright && playwright install chromium
python3 qa/qa_api.py      # негативные/авторизационные проверки backend (без учётных данных)
python3 qa/qa_web.py      # маркетинговый сайт: страницы, формы, меню, select, a11y
python3 qa/qa_admin.py    # админка: редиректы для анонима, страница входа
python3 qa/qa_mobile.py   # мобильное приложение на ЗАМОКАННОМ API (сеть не трогается, пароли не нужны)
python3 qa/qa_web_shop.py    # веб-версия: карта, вход, оплата, билет, заказы — на РЕАЛЬНОМ backend (seed-аккаунт, см. докстринг)
python3 qa/qa_web_oneclick.py # лента на главной: заказ в 1 клик (согласие, двойной клик, ошибки, выход) — на РЕАЛЬНОМ backend + витрина
python3 qa/qa_web_account.py  # сайт на РЕАЛЬНОМ backend: личный кабинет в шапке — у гостя лендинг-ссылки, у вошедшего только Витрина/Мои заказы без «Для бизнеса», выпадающее меню (фокус, стрелки, Esc), мобильная панель (закрытие по переходу, блокировка прокрутки), выход
python3 qa/qa_web_profile.py  # сайт на РЕАЛЬНОМ backend (одноразовый аккаунт qa.profile.*): профиль по образцу мобильного — пропуск, правка имени/email (валидация, занятый email), смена фото (реальная загрузка), тема система/светлая/тёмная (применяется до отрисовки, запоминается), окно помощи (mailto), выход, 375 px
python3 qa/qa_web_home.py  # сайт на РЕАЛЬНОМ backend: витрина /home — лента «Сейчас на витрине» + карта ниже, метки горячих с фото, жесты карты на прокручиваемой странице, геолокация (приглашение, отказ, «рядом со мной», вне города), лента сама плывёт (петля, паузы, переключатель в панели, reduced-motion), фильтры (категория, цена от/до с проверками), центровка цифр лендинга; iPhone 17/320/768/1024/1440 × Chromium/Firefox (WebKit на части машин зависает — QA_ENGINES=…,webkit включает явно)
python3 qa/qa_admin_photo.py  # админка: фото оффера (библиотека, своя загрузка, замена, снятие, ошибки) — на РЕАЛЬНОМ backend + seed
python3 qa/qa_mobile_offer.py  # мобильное приложение: экран оффера (фото, заведение, окно, остаток, панель покупки, маршрут, a11y) на ЗАМОКАННОМ API
python3 qa/qa_mobile_auth.py  # мобильное приложение: вход и регистрация на ЗАМОКАННОМ API (валидация, глазок, телефон, ошибки, чек-лист пароля, a11y)
python3 qa/qa_mobile_orders.py  # мобильное приложение: «где забрать» и маршрут, правило у «Оплатить», список заказов, помощь (письмо), офлайн-билет (холодный старт без сети, изоляция аккаунтов, очистка при выходе) на ЗАМОКАННОМ API
python3 qa/qa_mobile_welcome.py  # мобильное приложение: заставка со знаком, приветственная страница (арт-карта, пульсирующие «Войти»/«Создать аккаунт»), навигация, раскладка 320…768, reduce-motion, a11y на ЗАМОКАННОМ API
python3 qa/qa_mobile_map.py  # мобильное приложение: вкладка «Карта» — компактные карточки под картой, синхрон пин ↔ карточка, тап по соседней/выбранной, листание, клавиатура, список, геопозиция, пустое состояние, раскладка, a11y на ЗАМОКАННОМ API
python3 qa/qa_mobile_ribbon.py  # мобильное приложение: бегущая лента предложений над картой на «Сейчас» — едет без шва, замирает под пальцем/указателем/фокусом, нажатие, пауза на другой вкладке, reduce-motion, раскладка, a11y на ЗАМОКАННОМ API
python3 qa/qa_mobile_lab.py  # мобильное приложение: тестовый показ /ui-lab — новые элементы по мотивам 21st.dev (переключатель, радио-карточки, бейджи, пустые состояния, скелетоны, шторка со свайпом), без API
python3 qa/qa_mobile_real.py  # мобильное приложение на РЕАЛЬНОМ backend: вход демо-аккаунтом, оплата, настоящий QR, заказы, помощь, офлайн-перезапуск, выход (только localhost!)
python3 qa/qa_food_receipt_api.py  # РЕАЛЬНЫЙ backend (API): состав/аллергены/срок годности (валидация, правка, nearby) и ответ покупателя о получении (права, 409, гонка, спор, уведомления, счётчики)
python3 qa/qa_mobile_food.py  # мобильное приложение на моках: состав на экране оффера и в сводке оплаты, «Вы получили заказ?» (подтверждение, спор, ошибки, нет сети)
python3 qa/qa_web_food_receipt.py  # сайт на РЕАЛЬНОМ backend: состав на карте/оформлении/в ленте «в 1 клик» и подтверждение получения в билете
python3 qa/qa_admin_food.py  # админка на РЕАЛЬНОМ backend: «Состав и срок годности» (раскрывающийся список аллергенов, подстановка из прежних), пометка «нет состава», счётчик споров
python3 qa/qa_admin_address.py  # онбординг заведения: автоподсказки адреса (реальный Nominatim, нужен интернет) + выбор точки на карте; API замокан
python3 qa/qa_admin_hours.py  # РЕАЛЬНЫЙ backend + кабинет: часы работы заведения — оффер/шаблон обязан закончиться до закрытия (внутри/до открытия/после закрытия/выходной/за полночь), правка, конфликты при смене часов, автовыкладка пропускает невлезающее, редактор часов, быстрое создание режет конец по закрытию; часы подбираются от текущего момента
python3 qa/qa_admin_offer_manage.py  # РЕАЛЬНЫЙ backend + кабинет: ручная правка количества у выложенного оффера (не меньше проданного, раскуплен↔в продаже, гонка с заказом), удаление с витрины, отказ при заказах, удаление авто-оффера вносит дату в пропуски шаблона
python3 qa/qa_admin_sales.py  # РЕАЛЬНЫЙ backend + кабинет: график «Продажи» на «Аналитике» (7/14/30 дн., итоги = API, подробности дня, клавиатура, «когда покупают», пустое заведение, 375 px); данные: pnpm --filter @apetite/backend prisma:seed:sales-history
python3 qa/qa_mobile_breakfast.py  # мобильное приложение: вкладка «Завтрак» живёт с 00:00 до 10:00 (МСК), исчезает ровно в 10:00 по таймеру, откат на «Сейчас», пустое состояние, подпись режима скидки на экране оффера — на ЗАМОКАННОМ API
python3 qa/qa_admin_schedule.py  # РЕАЛЬНЫЙ backend + кабинет: режим скидки (растёт/фиксированная) и автовыкладка по шаблонам — материализация, идемпотентность, пропуск даты, пауза, удаление, права, валидация; форма «Выложить и повторять»
```

`qa_mobile.py` подставляет фейковые токены и отвечает на `/api/v1/**` из фикстур: так проверяются все экраны, пустые/ошибочные
состояния, оплата, отмена, 401→refresh, офлайн — детерминированно и без входа в аккаунт.
